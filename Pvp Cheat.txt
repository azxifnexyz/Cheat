-- ─── FORCE SPAWN LOGIC ────────────────────────────────────
local isSpamming = false

spawnBtn.MouseButton1Click:Connect(function()
    local name = toolBox.Text:lower()
    if name == "" then return end

    local char = LP.Character
    if not char then return end

    local RS = game:GetService("ReplicatedStorage")
    local toolsFolder = RS:FindFirstChild("Tools") or (RS:FindFirstChild("GunShop") and RS.GunShop:FindFirstChild("Tools"))

    if not toolsFolder then
        statusLbl.Text = "Tools folder not found!"
        return
    end

    local foundItem = nil
    for _, obj in ipairs(toolsFolder:GetDescendants()) do
        if (obj:IsA("Tool") or obj:IsA("Model")) and (obj.Name:lower() == name or obj.Name:lower():find(name)) then
            foundItem = obj
            break
        end
    end

    if foundItem then
        statusLbl.Text = "Found: " .. foundItem.Name .. ". Spawning..."
        
        local function cloneAndEquip()
            local clone = foundItem:Clone()
            
            -- Convert Model to Tool if needed
            if clone:IsA("Model") then
                local handle = clone:FindFirstChild("Handle")
                if handle then
                    local newTool = Instance.new("Tool")
                    newTool.Name = clone.Name
                    for _, child in ipairs(clone:GetChildren()) do
                        child.Parent = newTool
                    end
                    clone:Destroy()
                    clone = newTool
                end
            end

            if clone:IsA("Tool") then
                -- Force it directly into the player's hand
                clone.Parent = char
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum then
                    hum:EquipTool(clone)
                end
            end
        end

        -- Spawn 5 times rapidly to fight the anti-cheat
        for i = 1, 5 do
            cloneAndEquip()
            task.wait(0.1)
        end

        -- Anti-Delete Heartbeat Loop
        if not isSpamming then
            isSpamming = true
            task.spawn(function()
                for i = 1, 50 do
                    task.wait(0.1)
                    local hasTool = char:FindFirstChild(foundItem.Name) or LP.Backpack:FindFirstChild(foundItem.Name)
                    if not hasTool then
                        cloneAndEquip()
                    end
                end
                isSpamming = false
            end)
        end

        statusLbl.Text = "Spawned " .. foundItem.Name .. "!"
        spawnBtn.Text = "SPAWNED!"
        spawnBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 0)
        
        task.wait(1.5)
        spawnBtn.Text = "FORCE SPAWN"
        spawnBtn.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
    else
        statusLbl.Text = "Tool not found!"
        spawnBtn.Text = "NOT FOUND"
        spawnBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
        task.wait(1.5)
        spawnBtn.Text = "FORCE SPAWN"
        spawnBtn.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
    end
end)

-- ─── DRAG LOGIC ───────────────────────────────────────────
local dragging, dragStart, startPos
topBar.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        dragging = true; dragStart = i.Position; startPos = main.Position
    end
end)
UIS.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
end)
UIS.InputChanged:Connect(function(i)
    if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - dragStart
        main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
